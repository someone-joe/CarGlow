function Advance($event, $reason, $confirm) {
    $ab = @{event=$event; reason=$reason; confirm=$confirm} | ConvertTo-Json -Compress
    [System.IO.File]::WriteAllText('a.json', $ab)
    curl.exe -s -X POST "$base/admin-api/wash/order/$ORDERNO/advance" `
      -H "Content-Type: application/json" -H "Authorization: Bearer $adminToken" `
      --data-binary "@a.json"; echo ""
}
